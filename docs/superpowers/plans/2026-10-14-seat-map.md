## Review Focus

1. Two Residents tap the same seat in the same instant → exactly one gets it; the other gets "ที่นั่งไม่พร้อม กรุณาเลือกอีกครั้ง" and no row is left half-written (Task B2 tests it with the unique index and the hold rules, and Step 6 fires twelve real parallel connections at one seat).
2. A hold expires (60 s) while the dialog is open, or a Resident confirms a seat they never held → "หมดเวลาถือที่นั่ง กรุณาเลือกใหม่", no seat assigned (B2, C3).
3. A Resident who never scanned the QR tries to hold or be seated by Staff → refused with "ต้องสแกน QR ที่ห้องก่อน" (B2, B3).
4. A note containing a patient-looking number ("HN 12 34 56", "123456", "๑๒๓๔๕๖") → rejected by the database, not only by the screen (B2).
5. After `ends_at` (or after Admin closes the round) every write is refused and the map reads as ended; a session without a config row shows no map and changes nothing (B2, B4, C5).
6. A Resident must not learn which colleagues are absent or on urgent duty, nor who holds a pending hold (B4 tests the people list per role).
7. Staff who has not checked in, or a Resident, calling a Staff/Admin function → denied; `anon` can call nothing (B3, B5).

```js
const ROUND_TOKEN_UUID = "[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}";

// A scanned QR or a pasted link: accepts /attendance/<uuid> (with or without host, query or hash) or a bare uuid.
export function parseRoundTokenInput(value) {
  const text = String(value ?? "").trim();
  if (!text) return "";
  const bare = text.match(new RegExp(`^(${ROUND_TOKEN_UUID})$`, "i"));
  if (bare) return bare[1].toLowerCase();
  const link = text.match(new RegExp(`^(?:https?://[^/?#\\s]+)?/attendance/(${ROUND_TOKEN_UUID})/?(?:[?#].*)?$`, "i"));
  return link ? link[1].toLowerCase() : "";
}

// Facebook / LINE / Instagram in-app browsers usually refuse camera access.
export function isInAppBrowser(userAgent = typeof navigator === "undefined" ? "" : navigator.userAgent) {
  return /FBAN|FBAV|FB_IAB|Line\/|Instagram/i.test(String(userAgent || ""));
}
```

```jsx
import React, { useEffect, useRef, useState } from "react";
import { ScanIcon } from "../components/Icons";
import { isInAppBrowser, parseRoundTokenInput } from "../roundSchedule";

// stop() on a camera that is still starting leaves the stream running; always wait for `ready` first.
function stopScanner(scanner) {
  if (!scanner) return Promise.resolve();
  return scanner.ready.then(() => scanner.html5.stop()).catch(() => {}).finally(() => {
    try { scanner.html5.clear(); } catch { /* element already gone */ }
  });
}

export function RoundQrScanner({ onToken, disabled = false }) {
  const [camera, setCamera] = useState(false);
  const [manual, setManual] = useState("");
  const [error, setError] = useState("");
  const scannerRef = useRef(null);
  const stopping = useRef(Promise.resolve());
  const locked = useRef(false);
  const inApp = isInAppBrowser();

  function accept(value) {
    const token = parseRoundTokenInput(value);
    if (!token) { setError("QR นี้ไม่ใช่ QR เช็คชื่อประชุม กรุณาสแกน QR ที่แสดงในห้องประชุมหรือวางลิงก์เต็ม"); return false; }
    setError("");
    setCamera(false);
    setManual("");
    onToken(token);
    return true;
  }

  useEffect(() => {
    if (!camera) return undefined;
    let cancelled = false;
    stopping.current.then(() => import("html5-qrcode")).then(({ Html5Qrcode }) => {
      if (cancelled) return;
      const html5 = new Html5Qrcode("round-qr-reader", { verbose: false });
      const ready = html5.start({ facingMode: "environment" }, { fps: 10, qrbox: { width: 240, height: 240 } }, (decoded) => {
        if (locked.current) return;
        locked.current = true;
        const ok = accept(decoded);
        if (!ok) window.setTimeout(() => { locked.current = false; }, 1500);
      }, () => {});
      scannerRef.current = { html5, ready };
      ready.catch(() => {
        if (cancelled) return;
        setError("เปิดกล้องไม่สำเร็จ กรุณาอนุญาตการใช้กล้องของเว็บนี้ในการตั้งค่า (iPhone: การตั้งค่า > Safari > กล้อง) หรือวางลิงก์จาก QR ในช่องด้านล่าง");
        setCamera(false);
      });
    }).catch(() => { setError("เปิดตัวสแกน QR ไม่สำเร็จ กรุณาวางลิงก์จาก QR ในช่องด้านล่าง"); setCamera(false); });
    return () => { cancelled = true; stopping.current = stopScanner(scannerRef.current); scannerRef.current = null; locked.current = false; };
  }, [camera]);

  return <section className="round-scanner" aria-label="สแกน QR เช็คชื่อ">
    <button className={camera ? "secondary-button icon-button" : "primary-button icon-button"} type="button" disabled={disabled} onClick={() => { setError(""); setCamera((value) => !value); }}>
      <ScanIcon />{camera ? "ปิดกล้อง" : "สแกน QR เช็คชื่อ"}
    </button>
    {inApp && <p className="form-error" role="note">คุณเปิดเว็บนี้จากแอปแชต กล้องอาจถูกบล็อก กรุณาเปิดด้วย Safari หรือเปิดด้วย Chrome ของเครื่อง หรือวางลิงก์จาก QR ในช่องด้านล่าง</p>}
    {camera && <div id="round-qr-reader" className="qr-reader" />}
    <div className="manual-qr">
      <label>วางลิงก์จาก QR<input value={manual} onChange={(event) => setManual(event.target.value)} placeholder="วางลิงก์ /attendance/..." /></label>
      <button className="secondary-button" type="button" disabled={disabled || !manual.trim()} onClick={() => accept(manual)}>ใช้ลิงก์นี้</button>
    </div>
    {error && <p className="form-error" role="alert">{error}</p>}
  </section>;
}
```

```sql
--   'live'   = from starts_at to the effective end (ends_at, or earlier if an Admin closed the round)
--   'report' = from starts_at to ends_at + 48 h (urgent-duty reports and correction requests)
--   'read'   = anyone from starts_at - 1 h to ends_at + 48 h; an Admin at any time
create or replace function private.resident_seat_context(p_session_id uuid, p_phase text)
returns table (user_id uuid, role public.resident_system_role, layout_id uuid, starts_at timestamptz, ends_at timestamptz,
               effective_end timestamptz, report_until timestamptz, on_time_at timestamptz, now_ts timestamptz)

#variable_conflict use_column
declare
  v_a record;
  v_s record;

begin
  select a.user_id, a.role into v_a from private.resident_seat_actor() a;
  if v_a.user_id is null then raise exception 'Active account required'; end if;
  select * into v_s from private.resident_seat_session(p_session_id);
  if v_s.session_id is null then raise exception 'Seat map is not open for this session'; end if;

  if p_phase = 'report' and (v_now < v_s.starts_at or v_now > v_s.report_until) then raise exception 'Seat map is closed'; end if;
  if p_phase = 'read' and v_a.role <> 'admin' and (v_now < v_s.starts_at - interval '1 hour' or v_now > v_s.report_until) then raise exception 'Seat map is closed'; end if;
  return query select v_a.user_id, v_a.role, v_s.layout_id, v_s.starts_at, v_s.ends_at, v_s.effective_end, v_s.report_until, v_s.on_time_at, v_now;
```

```sql
-- ---------- Resident functions ----------

-- Take a seat in two steps: hold it for 60 s, then confirm. Two people tapping the same seat cannot both succeed.
create or replace function public.hold_resident_seat(p_session_id uuid, p_seat_code text)
returns timestamptz language plpgsql security definer set search_path = public, private, auth as $$
#variable_conflict use_column
declare
  c record;
  v_exp timestamptz;
begin
  select * into c from private.resident_seat_context(p_session_id, 'live');
  if c.role <> 'resident' then raise exception 'Active Resident account required'; end if;
  if not exists (select 1 from public.resident_round_attendance a where a.session_id = p_session_id and a.user_id = c.user_id) then
    raise exception 'Scan the room QR before choosing a seat';
  end if;
  if not exists (select 1 from public.resident_room_seats s where s.layout_id = c.layout_id and s.seat_code = p_seat_code) then
    raise exception 'Seat does not exist';
  end if;
  perform pg_advisory_xact_lock(hashtextextended(p_session_id::text || ':' || p_seat_code, 0));
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.expires_at <= c.now_ts;
  if exists (select 1 from public.resident_seat_assignments a where a.session_id = p_session_id and a.seat_code = p_seat_code and a.released_at is null)
     or exists (select 1 from public.resident_seat_holds h where h.session_id = p_session_id and h.seat_code = p_seat_code and h.user_id <> c.user_id) then
    raise exception 'Seat is not available';
  end if;
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.user_id = c.user_id;
  insert into public.resident_seat_holds(session_id, seat_code, user_id, expires_at)
  values (p_session_id, p_seat_code, c.user_id, c.now_ts + interval '60 seconds')
  returning expires_at into v_exp;
  perform private.resident_seat_log(p_session_id, c.user_id, c.user_id, 'hold', p_seat_code, '{}'::jsonb);
  return v_exp;
end;
$$;
revoke all on function public.hold_resident_seat(uuid, text) from public, anon;
grant execute on function public.hold_resident_seat(uuid, text) to authenticated;

create or replace function public.confirm_resident_seat(p_session_id uuid, p_seat_code text)
returns void language plpgsql security definer set search_path = public, private, auth as $$
#variable_conflict use_column
declare
  c record;
  v_old text;
begin
  select * into c from private.resident_seat_context(p_session_id, 'live');
  if c.role <> 'resident' then raise exception 'Active Resident account required'; end if;
  if not exists (select 1 from public.resident_round_attendance a where a.session_id = p_session_id and a.user_id = c.user_id) then
    raise exception 'Scan the room QR before choosing a seat';
  end if;
  perform pg_advisory_xact_lock(hashtextextended(p_session_id::text || ':' || p_seat_code, 0));
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.expires_at <= c.now_ts;
  if not exists (select 1 from public.resident_seat_holds h where h.session_id = p_session_id and h.seat_code = p_seat_code and h.user_id = c.user_id) then
    raise exception 'Seat hold expired. Choose the seat again';
  end if;
  if exists (select 1 from public.resident_seat_assignments a where a.session_id = p_session_id and a.seat_code = p_seat_code and a.released_at is null) then
    raise exception 'Seat is not available';
  end if;
  begin
    update public.resident_seat_assignments a set released_at = c.now_ts, release_reason = 'moved'
    where a.session_id = p_session_id and a.user_id = c.user_id and a.released_at is null
    returning a.seat_code into v_old;
    insert into public.resident_seat_assignments(session_id, user_id, seat_code, seated_at, assigned_by)
    values (p_session_id, c.user_id, p_seat_code, c.now_ts, c.user_id);
  exception when unique_violation then
    raise exception 'Seat is not available';
  end;
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.user_id = c.user_id;
  delete from public.resident_seat_status s where s.session_id = p_session_id and s.user_id = c.user_id;
  perform private.resident_seat_log(p_session_id, c.user_id, c.user_id, case when v_old is null then 'confirm' else 'move' end, p_seat_code,
    case when v_old is null then '{}'::jsonb else jsonb_build_object('from', v_old) end);
end;
$$;
revoke all on function public.confirm_resident_seat(uuid, text) from public, anon;
grant execute on function public.confirm_resident_seat(uuid, text) to authenticated;

-- Urgent duty (surgery, emergency, on call): no patient data, no QR scan needed, reportable until 48 h after the session.
create or replace function public.report_urgent_duty(p_session_id uuid, p_reason text, p_note text default null)
returns uuid language plpgsql security definer set search_path = public, private, auth as $$
#variable_conflict use_column
declare
  c record;
  v_note text := nullif(btrim(coalesce(p_note, '')), '');
  v_seat text;
  v_id uuid;
begin
  select * into c from private.resident_seat_context(p_session_id, 'report');
  if c.role <> 'resident' then raise exception 'Active Resident account required'; end if;
  if p_reason is null or p_reason not in ('ผ่าตัด', 'ผู้ป่วยฉุกเฉิน', 'อยู่เวร', 'อื่น ๆ') then raise exception 'Reason is not valid'; end if;
  if char_length(coalesce(v_note, '')) > 160 then raise exception 'Note is too long'; end if;
  if not private.resident_seat_text_ok(v_note) then raise exception 'Remove numbers that could identify a patient'; end if;
  begin
    insert into public.resident_seat_requests(session_id, user_id, kind, reason, note)
    values (p_session_id, c.user_id, 'urgent', p_reason, v_note) returning id into v_id;
  exception when unique_violation then
    raise exception 'You already reported urgent duty for this session';
  end;
```

```sql
-- Staff who checked in (and Admin) seat or move a Resident who ALREADY scanned the room QR. No swaps: an occupied seat is refused.
create or replace function public.staff_set_resident_seat(p_session_id uuid, p_resident_id uuid, p_seat_code text)
returns void language plpgsql security definer set search_path = public, private, auth as $$
#variable_conflict use_column
declare
  c record;
  v_old text;
begin
  select * into c from private.resident_seat_context(p_session_id, 'live');
  if not (select private.resident_can_manage_seat_map(p_session_id)) then raise exception 'Staff who checked in, or Admin, required'; end if;
  if not exists (
    select 1 from public.resident_user_roles r join public.resident_profiles p on p.user_id = r.user_id
    where r.user_id = p_resident_id and r.role = 'resident' and r.active and p.active
  ) then raise exception 'Resident not found'; end if;
  if not exists (select 1 from public.resident_round_attendance a where a.session_id = p_session_id and a.user_id = p_resident_id) then
    raise exception 'This Resident has not scanned the room QR. Ask them to send a correction request';
  end if;
  if not exists (select 1 from public.resident_room_seats s where s.layout_id = c.layout_id and s.seat_code = p_seat_code) then
    raise exception 'Seat does not exist';
  end if;
  perform pg_advisory_xact_lock(hashtextextended(p_session_id::text || ':' || p_seat_code, 0));
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.expires_at <= c.now_ts;
  if exists (select 1 from public.resident_seat_assignments a where a.session_id = p_session_id and a.seat_code = p_seat_code and a.released_at is null)
     or exists (select 1 from public.resident_seat_holds h where h.session_id = p_session_id and h.seat_code = p_seat_code and h.user_id <> p_resident_id) then
    raise exception 'Seat is not available';
  end if;
  begin
    update public.resident_seat_assignments a set released_at = c.now_ts, release_reason = 'staff'
    where a.session_id = p_session_id and a.user_id = p_resident_id and a.released_at is null
    returning a.seat_code into v_old;
    insert into public.resident_seat_assignments(session_id, user_id, seat_code, seated_at, assigned_by)
    values (p_session_id, p_resident_id, p_seat_code, c.now_ts, c.user_id);
  exception when unique_violation then
    raise exception 'Seat is not available';
  end;
  delete from public.resident_seat_holds h where h.session_id = p_session_id and h.user_id = p_resident_id;
  delete from public.resident_seat_status s where s.session_id = p_session_id and s.user_id = p_resident_id;
  perform private.resident_seat_log(p_session_id, c.user_id, p_resident_id, 'staff_set', p_seat_code,
    case when v_old is null then '{}'::jsonb else jsonb_build_object('from', v_old) end);
end;
$$;

  if not (select private.resident_can_configure_seat_map()) then raise exception 'Active Admin account required'; end if;

  from public.resident_round_seat_config cfg join public.resident_round_sessions s2 on s2.id = cfg.session_id
  where cfg.session_id <> p_session_id and s2.starts_at < v_s.starts_at
  order by s2.starts_at desc limit 1;

  v_layout := coalesce(p_layout_id, v_exist.layout_id, v_prev.layout_id,
                       (select l.id from public.resident_room_layouts l where l.active order by l.name limit 1));

    raise exception 'Layout not found';

    if p_on_time_at < v_s.starts_at or p_on_time_at > v_s.ends_at then raise exception 'On-time must be inside the session window'; end if;
    v_on := p_on_time_at;
  elsif v_exist.on_time_at is not null then
    v_on := v_exist.on_time_at;
  else
    v_on := case when v_prev.on_time_at is null then null else
      ((v_s.starts_at at time zone 'Asia/Bangkok')::date + (v_prev.on_time_at at time zone 'Asia/Bangkok')::time) at time zone 'Asia/Bangkok' end;
    if v_on is null or v_on < v_s.starts_at or v_on > v_s.ends_at then v_on := least(v_s.starts_at + interval '15 minutes', v_s.ends_at); end if;
  end if;

  return query select p_session_id, v_layout, v_on;
end;
$$;
```

```sql
-- ---------- Read functions ----------

-- Residents cannot read resident_round_sessions (Admin-only RLS), so the sessions that have a seat map come from here.
create or replace function public.list_resident_seat_sessions()
returns table (session_id uuid, meeting_date date, starts_at timestamptz, ends_at timestamptz, effective_end timestamptz, on_time_at timestamptz)
language plpgsql stable security definer set search_path = public, private, auth as $$
#variable_conflict use_column
begin
  if not exists (select 1 from private.resident_seat_actor()) then raise exception 'Active account required'; end if;
  return query
  select s.id, s.meeting_date, s.starts_at, s.ends_at, least(s.ends_at, coalesce(s.closed_at, s.ends_at)), c.on_time_at
  from public.resident_round_sessions s
  join public.resident_round_seat_config c on c.session_id = s.id
  where s.cancelled_at is null
    and clock_timestamp() between s.starts_at - interval '1 hour' and s.ends_at + interval '48 hours'
  order by s.starts_at, s.id;
end;
$$;
revoke all on function public.list_resident_seat_sessions() from public, anon;
grant execute on function public.list_resident_seat_sessions() to authenticated;

-- The whole picture for one session in one call. Names are visible to every signed-in user (decision B); absent people,
-- urgent-duty status and other people's entry times are visible only to Staff and Admin; a hold shows as a flag, never as a person.
create or replace function public.get_resident_seat_map(p_session_id uuid)
returns jsonb language plpgsql security definer set search_path = public, private, auth as $$
#variable_conflict use_column
declare
  c record;
  v_priv boolean;
  v_att timestamptz;
  v_seat text;
  v_hold record;
  v_urgent text;
  v_reserve boolean;
  v_status text;
  v_seats jsonb;
  v_people jsonb;
  v_counts jsonb;
begin
  select * into c from private.resident_seat_context(p_session_id, 'read');
  v_priv := c.role in ('staff', 'admin');
  select t.checked_in_at into v_att from public.resident_round_attendance t where t.session_id = p_session_id and t.user_id = c.user_id;
  select a.seat_code into v_seat from public.resident_seat_assignments a where a.session_id = p_session_id and a.user_id = c.user_id and a.released_at is null;
  select h.seat_code, h.expires_at into v_hold from public.resident_seat_holds h where h.session_id = p_session_id and h.user_id = c.user_id and h.expires_at > c.now_ts;
  select q.state into v_urgent from public.resident_seat_requests q
    where q.session_id = p_session_id and q.user_id = c.user_id and q.kind = 'urgent' and q.state <> 'rejected' order by q.created_at desc limit 1;
  v_reserve := exists (select 1 from public.resident_seat_status s where s.session_id = p_session_id and s.user_id = c.user_id);
  v_status := case when v_seat is not null then 'seat' when v_reserve then 'reserve'
                   when v_att is not null and c.role = 'staff' then 'staff' when v_att is not null then 'room'
                   when v_urgent is not null then 'urgent' else 'absent' end;

  select coalesce(jsonb_agg(jsonb_build_object(
      'seat_code', s.seat_code, 'zone', s.zone, 'row_no', s.row_no, 'col_no', s.col_no,
      'held', (h.seat_code is not null),
      'occupant', case when a.user_id is null then null else jsonb_build_object(
          'user_id', a.user_id, 'name', p.full_name, 'pgy', p.pgy, 'role', r.role, 'seated_at', a.seated_at) end
    ) order by s.zone, s.row_no, s.col_no), '[]'::jsonb)
  into v_seats
  from public.resident_room_seats s
  left join public.resident_seat_assignments a on a.session_id = p_session_id and a.seat_code = s.seat_code and a.released_at is null
  left join public.resident_profiles p on p.user_id = a.user_id
  left join public.resident_user_roles r on r.user_id = a.user_id
  left join public.resident_seat_holds h on h.session_id = p_session_id and h.seat_code = s.seat_code and h.expires_at > c.now_ts
  where s.layout_id = c.layout_id;

  with base as (
    select p.user_id, p.full_name, p.pgy, r.role::text as role, t.checked_in_at, a.seat_code, (st.user_id is not null) as reserve,
           (select q.state from public.resident_seat_requests q
             where q.session_id = p_session_id and q.user_id = p.user_id and q.kind = 'urgent' and q.state <> 'rejected'
             order by q.created_at desc limit 1) as urgent_state
    from public.resident_profiles p
    join public.resident_user_roles r on r.user_id = p.user_id and r.active and r.role in ('resident', 'staff')
    left join public.resident_round_attendance t on t.session_id = p_session_id and t.user_id = p.user_id
    left join public.resident_seat_assignments a on a.session_id = p_session_id and a.user_id = p.user_id and a.released_at is null
    left join public.resident_seat_status st on st.session_id = p_session_id and st.user_id = p.user_id
    where p.active
  ), shown as (
    select b.*, case
        when b.seat_code is not null then 'seat'
        when b.reserve then 'reserve'
        when b.checked_in_at is not null and b.role = 'staff' then 'staff'
        when b.checked_in_at is not null then 'room'
        when b.urgent_state is not null then 'urgent'
        else 'absent' end as status
    from base b
  )
  select coalesce(jsonb_agg(jsonb_build_object(
      'user_id', s.user_id, 'name', s.full_name, 'pgy', s.pgy, 'role', s.role, 'status', s.status, 'seat_code', s.seat_code,
      'checked_in_at', case when v_priv or s.user_id = c.user_id then s.checked_in_at end,
      'on_time', case when (v_priv or s.user_id = c.user_id) and s.checked_in_at is not null then s.checked_in_at < c.on_time_at end
    ) order by s.full_name, s.user_id), '[]'::jsonb)
  into v_people
  from shown s
  where v_priv or s.status in ('seat', 'reserve', 'room', 'staff');

  select jsonb_build_object(
    'seats_total', (select count(*) from public.resident_room_seats s where s.layout_id = c.layout_id),
    'seats_taken', (select count(*) from public.resident_seat_assignments a where a.session_id = p_session_id and a.released_at is null),
    'residents_total', (select count(*) from public.resident_user_roles r join public.resident_profiles p on p.user_id = r.user_id where r.role = 'resident' and r.active and p.active),
    'residents_in_room', (select count(*) from public.resident_round_attendance t where t.session_id = p_session_id and t.role_at_check_in = 'resident')
  ) into v_counts;

  return jsonb_build_object(
    'session', jsonb_build_object('id', p_session_id, 'starts_at', c.starts_at, 'ends_at', c.ends_at, 'effective_end', c.effective_end,
                                  'on_time_at', c.on_time_at, 'layout_id', c.layout_id, 'now', c.now_ts,
                                  'ended', c.now_ts > c.effective_end, 'open', c.now_ts between c.starts_at and c.effective_end),
    'me', jsonb_build_object('user_id', c.user_id, 'role', c.role, 'in_room', v_att is not null, 'checked_in_at', v_att,
                             'on_time', case when v_att is null then null else v_att < c.on_time_at end,
                             'seat', v_seat, 'reserve', v_reserve,
                             'hold', case when v_hold.seat_code is null then null else jsonb_build_object('seat_code', v_hold.seat_code, 'expires_at', v_hold.expires_at) end,
                             'urgent_state', v_urgent, 'status', v_status),
    'seats', v_seats, 'people', v_people, 'counts', v_counts);
end;
$$;
revoke all on function public.get_resident_seat_map(uuid) from public, anon;

-- On-screen report for the Admin: who came before / after the on-time time, requests, and the full event log.

      'correction_rejected', count(*) filter (where q.kind = 'correction' and q.state = 'rejected'))
    from public.resident_seat_requests q where q.session_id = p_session_id)
  || jsonb_build_object(
      'seated', (select count(*) from public.resident_seat_assignments a where a.session_id = p_session_id and a.released_at is null),
      'reserve', (select count(*) from public.resident_seat_status s where s.session_id = p_session_id),
      'certifications', (select count(*) from public.resident_seat_events e where e.session_id = p_session_id and e.kind = 'certify'));

      'on_time', case when t.checked_in_at is null then null else t.checked_in_at < c.on_time_at end,
      'seat_code', a.seat_code, 'reserve', (st.user_id is not null),

    ) order by p.full_name, p.user_id), '[]'::jsonb)
  into v_people
  from public.resident_profiles p
  join public.resident_user_roles r on r.user_id = p.user_id and r.active and r.role in ('resident', 'staff')
  left join public.resident_round_attendance t on t.session_id = p_session_id and t.user_id = p.user_id
  left join public.resident_seat_assignments a on a.session_id = p_session_id and a.user_id = p.user_id and a.released_at is null
  left join public.resident_seat_status st on st.session_id = p_session_id and st.user_id = p.user_id
  where p.active;

  select coalesce(jsonb_agg(jsonb_build_object(
      'id', q.id, 'kind', q.kind, 'reason', q.reason, 'note', q.note, 'state', q.state, 'user_id', q.user_id, 'name', up.full_name,

  left join public.resident_profiles rp on rp.user_id = q.reviewed_by

  where e.session_id = p_session_id;

    'counts', v_counts, 'people', v_people, 'requests', v_requests, 'events', v_events);
end;
$$;
```

```sql
do $$
declare
  v_tables text[] := array['resident_room_layouts','resident_room_seats','resident_round_seat_config','resident_seat_assignments','resident_seat_holds','resident_seat_status','resident_seat_requests','resident_seat_events'];

  foreach v_t in array v_tables loop
    if to_regclass('public.' || v_t) is null then raise exception 'Missing table %', v_t; end if;
    if not (select relrowsecurity from pg_class where oid = ('public.' || v_t)::regclass) then raise exception 'RLS is off on %', v_t; end if;
    if has_table_privilege('anon', 'public.' || v_t, 'select') or has_table_privilege('anon', 'public.' || v_t, 'insert')

      raise exception 'Signed-in users can read % directly', v_t; end if;
  end loop;
  foreach v_f in array v_fns loop

    if not has_function_privilege('authenticated', v_f, 'execute') then raise exception 'authenticated cannot run %', v_f; end if;
```

```js
export function seatsByZone(seats = []) {
```

```jsx
export function HoldDialog({ seat, from, expiresAt, offsetMs, busy, error, onConfirm, onCancel }) {
```

```jsx
import React, { useCallback, useEffect, useRef, useState } from "react";
import {
  ENDED_NOTE, MODERATOR_ZONE, SEAT_POLL_MS, ZONE_INFO, bangkokHm, canManageSeats, canPickSeat, pickDefaultSession, seatCell, seatErrorMessage, serverOffsetMs, shouldPoll,
} from "../seatMap";
import {
  certifySeatMap, confirmSeat, holdSeat, listSeatSessions, loadMySeatEvents, loadSeatMap, moveToReserveArea, releaseSeat,
  reportUrgentDuty, requestSeatCorrection, staffReleaseSeat, staffSetSeat,
} from "../seatMapApi";
import { CorrectionDialog, DesktopRoom, HistoryTab, HoldDialog, ManageDialog, MiniRoom, PeopleTab, PersonalCard, UrgentDialog, ZoneGrid } from "./SeatMapViews";

const thaiDate = (iso) => new Intl.DateTimeFormat("th-TH", { dateStyle: "full", timeZone: "Asia/Bangkok" }).format(new Date(iso));
const TABS = [["map", "ผังที่นั่ง"], ["people", "ผู้เข้าร่วม"], ["history", "ประวัติของฉัน"]];

// Shown on the "เช็คชื่อประชุม" page for every session that has a seat map; renders nothing when there is none.
export function SeatMapPanel({ user, refreshKey = "" }) {
  const [sessions, setSessions] = useState(null);
  const [sessionId, setSessionId] = useState("");
  const [map, setMap] = useState(null);
  const [offsetMs, setOffsetMs] = useState(0);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");
  const [dialogError, setDialogError] = useState("");
  const [busy, setBusy] = useState(false);
  const [tab, setTab] = useState("map");
  const [zone, setZone] = useState("L");
  const [dialog, setDialog] = useState(null);
  const [history, setHistory] = useState(null);
  const [query, setQuery] = useState("");
  const [status, setStatus] = useState("all");
  const mounted = useRef(true);
  const ticket = useRef(0);
  useEffect(() => { mounted.current = true; return () => { mounted.current = false; }; }, []);

  const loadSessions = useCallback(async () => {
    try {
      const list = await listSeatSessions();
      if (!mounted.current) return;
      setSessions(list);
      setSessionId((previous) => (previous && list.some((item) => item.session_id === previous) ? previous : pickDefaultSession(list, Date.now())));
    } catch (nextError) {
      if (mounted.current) { setSessions([]); setError(seatErrorMessage(nextError)); }
    }
  }, []);

  const refresh = useCallback(async (id) => {
    if (!id) return;
    const mine = ++ticket.current;
    try {
      const requestedAt = Date.now();
      const next = await loadSeatMap(id);
      if (!mounted.current || mine !== ticket.current) return;
      setMap(next);
      setOffsetMs(serverOffsetMs(next, requestedAt));
      setError("");
    } catch (nextError) {
      if (mounted.current && mine === ticket.current) setError(seatErrorMessage(nextError));
    }
  }, []);

  useEffect(() => { loadSessions(); }, [refreshKey, loadSessions]);
  useEffect(() => { setMap(null); setHistory(null); refresh(sessionId); }, [sessionId, refresh]);
  useEffect(() => { if (sessionId) refresh(sessionId); }, [refreshKey]);

  const dialogOpen = Boolean(dialog);
  const ended = Boolean(map?.session?.ended);
  useEffect(() => {
    if (!sessionId) return undefined;
    const tick = () => { if (shouldPoll({ hidden: document.hidden, dialogOpen, ended })) refresh(sessionId); };
    const timer = window.setInterval(tick, SEAT_POLL_MS);
    const onVisible = () => { if (!document.hidden) tick(); };
    document.addEventListener("visibilitychange", onVisible);
    return () => { window.clearInterval(timer); document.removeEventListener("visibilitychange", onVisible); };
  }, [sessionId, dialogOpen, ended, refresh]);

  useEffect(() => {
    if (tab !== "history" || !sessionId) return;
    loadMySeatEvents(sessionId).then((rows) => { if (mounted.current) setHistory(rows); }).catch((nextError) => { if (mounted.current) setError(seatErrorMessage(nextError)); });
  }, [tab, sessionId, map?.me?.seat, map?.me?.status]);

  // Every action: call, reload the map, close the dialog on success, keep it open and explain on failure.
  async function act(run, success) {
    setBusy(true); setNotice(""); setError(""); setDialogError("");
    try {
      await run();
      if (!mounted.current) return;
      setDialog(null); setNotice(success);
      await refresh(sessionId);
    } catch (nextError) {
      if (!mounted.current) return;
      const message = seatErrorMessage(nextError);
      if (dialog) setDialogError(message); else setError(message);
      await refresh(sessionId);
    } finally { if (mounted.current) setBusy(false); }
  }

  async function onSeat(seat) {
    if (!map) return;
    if (canManageSeats(map)) { setDialogError(""); setDialog({ type: "manage", seat }); return; }
    const cell = seatCell(seat, map);
    if (!canPickSeat(map) || !(cell.state === "free" || cell.state === "myhold")) return;
    setBusy(true); setError(""); setNotice("");
    try {
      const expiresAt = await holdSeat(sessionId, seat.seat_code);
      if (mounted.current) { setDialogError(""); setDialog({ type: "hold", seat, expiresAt }); }
    } catch (nextError) {
      if (mounted.current) setError(seatErrorMessage(nextError));
      await refresh(sessionId);
    } finally { if (mounted.current) setBusy(false); }
  }

  if (sessions === null || !sessions.length) return error && sessions ? <p className="form-error" role="alert">{error}</p> : null;
  const session = sessions.find((item) => item.session_id === sessionId) || sessions[0];
  const zoneName = zone === "M" ? MODERATOR_ZONE.name : ZONE_INFO.find((item) => item.zone === zone)?.name;
  return <section className="resident-panel seatmap" aria-labelledby="seatmap-title">
    <div className="seatmap-head">
      <h2 id="seatmap-title">ผังที่นั่งประชุม</h2>
      {sessions.length > 1
        ? <label>รอบ<select value={sessionId} onChange={(event) => setSessionId(event.target.value)}>{sessions.map((item) => <option key={item.session_id} value={item.session_id}>{thaiDate(item.meeting_date)} · {bangkokHm(item.starts_at)}–{bangkokHm(item.ends_at)} น.</option>)}</select></label>
        : <span>{thaiDate(session.meeting_date)} · {bangkokHm(session.starts_at)}–{bangkokHm(session.ends_at)} น.</span>}
    </div>
    {error && <p className="form-error" role="alert">{error}</p>}
    {notice && <p className="form-success" role="status">{notice}</p>}
    {!map ? <p role="status">กำลังโหลดผังที่นั่ง…</p> : <>
      {map.session.ended && <p className="seatmap-ended" role="note">การประชุมนี้จบแล้ว · {ENDED_NOTE}</p>}
      <PersonalCard map={map} busy={busy}
        onRelease={() => act(() => releaseSeat(sessionId), "คืนที่นั่งแล้ว")}
        onReserve={() => act(() => moveToReserveArea(sessionId), "ย้ายไปที่นั่งเสริมแล้ว")}
        onUrgent={() => { setDialogError(""); setDialog({ type: "urgent" }); }}
        onCorrection={() => { setDialogError(""); setDialog({ type: "correction" }); }}
        onCertify={() => act(() => certifySeatMap(sessionId), "บันทึกการรับรองผังแล้ว")} />
      <div className="seatmap-tabs" role="tablist" aria-label="หัวข้อผังที่นั่ง">
        {TABS.map(([id, label]) => <button key={id} type="button" role="tab" aria-selected={tab === id} onClick={() => setTab(id)}>{label}</button>)}
      </div>
      {tab === "map" && <>
        <div className="seatmap-mobile-view">
          <MiniRoom map={map} activeZone={zone} onZone={setZone} />
          <div className="seatmap-zone-detail"><h3>{zoneName} <span className="muted">· {canManageSeats(map) ? "แตะที่นั่งเพื่อจัดที่" : "เลือกที่นั่ง"}</span></h3><ZoneGrid zone={zone} map={map} onSeat={onSeat} /></div>
        </div>
        <div className="seatmap-desktop-view"><DesktopRoom map={map} onSeat={onSeat} /></div>
      </>}
      {tab === "people" && <PeopleTab map={map} query={query} onQuery={setQuery} status={status} onStatus={setStatus} />}
      {tab === "history" && <HistoryTab events={history} />}
    </>}
    {dialog?.type === "hold" && <HoldDialog seat={dialog.seat} from={map?.me?.seat || ""} expiresAt={dialog.expiresAt} offsetMs={offsetMs} busy={busy} error={dialogError}
      onConfirm={() => act(() => confirmSeat(sessionId, dialog.seat.seat_code), "บันทึกที่นั่งแล้ว")} onCancel={() => { setDialog(null); refresh(sessionId); }} />}
    {dialog?.type === "manage" && <ManageDialog seat={dialog.seat} map={map} busy={busy} error={dialogError}
      onSet={(residentId) => act(() => staffSetSeat(sessionId, residentId, dialog.seat.seat_code), "จัดที่นั่งแล้ว")}
      onRelease={(residentId) => act(() => staffReleaseSeat(sessionId, residentId), "คืนที่นั่งแล้ว")} onClose={() => setDialog(null)} />}
    {dialog?.type === "urgent" && <UrgentDialog busy={busy} error={dialogError}
      onSubmit={({ reason, note }) => act(() => reportUrgentDuty(sessionId, reason, note), "บันทึกภารกิจจำเป็นแล้ว · รอผู้ดูแลตรวจสอบ")} onCancel={() => setDialog(null)} />}
    {dialog?.type === "correction" && <CorrectionDialog busy={busy} error={dialogError}
      onSubmit={({ note }) => act(() => requestSeatCorrection(sessionId, note), "ส่งคำขอแก้ไขแล้ว")} onCancel={() => setDialog(null)} />}
  </section>;
}
```
